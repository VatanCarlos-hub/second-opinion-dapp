import "jsr:@supabase/functions-js/edge-runtime.d.ts"; import { createClient } from "jsr:@supabase/supabase-js@2"; import { createClient as glCreateClient, createAccount } from "npm:genlayer-js"; import { studionet } from "npm:genlayer-js/chains"; import { createHmac } from "node:crypto"; const CONTRACT = Deno.env.get("GL_CONTRACT")!; const MASTER_SEED = Deno.env.get("MASTER_SEED")!; const cors = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", }; function json(body: unknown, status = 200): Response { return new Response(JSON.stringify(body), { status, headers: { ...cors, "Content-Type": "application/json" }, }); } function derivePk(email: string): string { const norm = email.trim().toLowerCase(); return "0x" + createHmac("sha256", MASTER_SEED).update("gl-user:" + norm).digest("hex"); } function parseVerdict(raw: string) { let vStr = String(raw || "").replace(/^"|"$/g, ""); let therapy: unknown = null; try { const p = JSON.parse(raw); if (p.verdict) vStr = p.verdict; if (p.therapy) { try { therapy = typeof p.therapy === "string" ? JSON.parse(p.therapy) : p.therapy; } catch { therapy = p.therapy; } } } catch { // roher String, kein JSON } const parts = vStr.split(" | "); const rec = (parts[parts.length - 1] || "").trim(); const urgency = ["URGENT", "SOON", "MONITOR"].includes(rec) ? rec : "MONITOR"; const text = parts.length > 1 ? parts.slice(0, parts.length - 1).join(" | ") : vStr; return { urgency, text, therapy, verdict: vStr }; } Deno.serve(async (req) => { if (req.method === "OPTIONS") { return new Response("ok", { headers: cors }); } if (req.method !== "POST") { return json({ error: "Method not allowed" }, 405); } try { const jwt = (req.headers.get("Authorization") || "").replace("Bearer ", ""); const admin = createClient( Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")! ); const { data: u, error: uErr } = await admin.auth.getUser(jwt); if (uErr || !u || !u.user || !u.user.email) { return json({ error: "Nicht eingeloggt." }, 401); } const email = u.user.email; const b = await req.json().catch(() => ({})); const action = b.action || "submit"; const account = createAccount(derivePk(email)); // ---------- PHASE 1: nur einreichen, sofort zurueckkehren ---------- if (action === "submit") { if (!b.symptoms || !String(b.age || "").trim()) { return json({ error: "Alter und Symptome sind noetig." }, 400); } const client = glCreateClient({ chain: studionet, account }); const txHash = await client.writeContract({ address: CONTRACT, functionName: "submit_consultation", args: [ b.symptoms, String(b.age), b.gender || "", String(b.weight || "0"), String(b.height || "0"), b.allergies || "", b.preConditions || "", b.docText || "", ], value: 0n, }); return json({ txHash: txHash, glAddress: account.address }); } // ---------- PHASE 2: Verdikt lesen und speichern (kurz) ---------- if (action === "result") { const reader = glCreateClient({ chain: studionet }); const raw = await reader.readContract({ address: CONTRACT, functionName: "get_verdict", args: [account.address.toLowerCase()], }); const parsed = parseVerdict(String(raw || "")); if (!parsed.verdict || parsed.verdict.length < 5) { return json({ error: "Verdikt noch nicht verfuegbar." }, 404); } await admin.from("consultations").insert({ email: email, gl_address: account.address, urgency: parsed.urgency, verdict: parsed.verdict, tx_hash: b.txHash || null, }); return json({ urgency: parsed.urgency, text: parsed.text, therapy: parsed.therapy, }); } return json({ error: "Unbekannte Aktion." }, 400); } catch (e) { return json({ error: (e as Error).message || "GenLayer-Fehler." }, 502); } });
